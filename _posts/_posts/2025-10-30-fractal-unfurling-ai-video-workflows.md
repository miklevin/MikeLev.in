```python
# The "secret sauce" for our pd.concat()-for-video workflow.

from moviepy.editor import VideoFileClip, concatenate_videoclips
from pathlib import Path

def simple_concatenate(job_name: str, clip_list: list, output_filename: str) -> str:
    """
    The "baby-step" win. Takes a list of video file paths and joins them.
    """
    print(f"🎬 Starting video concatenation job: {job_name}")
    
    # --- 1. Define Paths ---
    # We assume clips are in 'Notebooks/clips/' and output goes to 'Notebooks/deliverables/'
    script_location = Path(__file__).resolve().parent # Notebooks/imports
    notebook_root = script_location.parent          # Notebooks/
    
    clips_dir = notebook_root / "clips"
    output_dir = notebook_root / "deliverables" / job_name
    output_dir.mkdir(parents=True, exist_ok=True)
    
    output_path = output_dir / output_filename
    
    # --- 2. Load Clips ---
    video_clips = []
    for clip_file in clip_list:
        clip_path = clips_dir / clip_file
        if not clip_path.exists():
            print(f"⚠️ Warning: Clip not found, skipping: {clip_path}")
            continue
            
        print(f"  -> Loading clip: {clip_file}")
        try:
            # Load the clip
            video_clips.append(VideoFileClip(str(clip_path)))
        except Exception as e:
            print(f"❌ Error loading clip {clip_file}: {e}")
            # Clean up any clips already loaded to avoid file handle issues
            for clip in video_clips:
                clip.close()
            return None

    if not video_clips:
        print("❌ No valid video clips were loaded. Aborting.")
        return None

    # --- 3. Concatenate (The "pd.concat()" part) ---
    try:
        print(f"🎞️ Concatenating {len(video_clips)} clips...")
        final_clip = concatenate_videoclips(video_clips)
        
        # Write the final video file
        # We use 'libx264' for broad compatibility (MP4)
        print(f"💾 Writing final video to: {output_path}")
        final_clip.write_videofile(str(output_path), codec='libx264', audio_codec='aac')
        
    except Exception as e:
        print(f"❌ Error during concatenation or writing: {e}")
        return None
    finally:
        # --- 4. Clean up file handles ---
        print("🧹 Cleaning up clip file handles...")
        for clip in video_clips:
            clip.close() # CRITICAL: Close all VideoFileClip handles
        if 'final_clip' in locals():
            final_clip.close()
            
    print(f"✅ Success! Video saved to: {output_path}")
    return str(output_path)
```
